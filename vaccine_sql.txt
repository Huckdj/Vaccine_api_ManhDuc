CREATE FUNCTION [dbo].[GetErrorMessage] (@ErrorCode INT)
RETURNS NVARCHAR(255)
AS
BEGIN
    DECLARE @ErrorMessage NVARCHAR(255);

    -- Truy vấn để lấy thông điệp lỗi từ bảng ErrorCodes
    SELECT @ErrorMessage = ErrorMessage
    FROM ErrorCodes
    WHERE ErrorCode = @ErrorCode;

    -- Nếu không tìm thấy mã lỗi, trả về một thông báo mặc định
    IF @ErrorMessage IS NULL
    BEGIN
        SET @ErrorMessage = 'Unknown error code';
    END

    RETURN @ErrorMessage;
END
GO

CREATE PROC [dbo].[ExecAccount]

	@Password nvarchar(255) = '135790@Duc',

AS
BEGIN

	BEGIN
		select 400 ErrorCode, dbo.GetErrorMessage(1) AS ErrorMessage
		goto THOAT
	END


	DECLARE @Sql NVARCHAR(MAX) = ''
	IF @Type = 'Auth-Login'
	BEGIN
		IF (select count(*) from VaccineCategory.dbo.Account where Email = @UserAuth or SDT = @UserAuth) >= 1
		BEGIN
			IF(select count(*) from VaccineCategory.dbo.Account where (Email = @UserAuth OR SDT = @UserAuth) AND Password = @Password AND isActive = 1 AND isDeleted = 0) >= 1
			BEGIN
				SELECT TOP(1)  ID,Email ,Admin, 0 ErrorCode, dbo.GetErrorMessage(4) AS ErrorMessage
				FROM VaccineCategory.dbo.Account
				WHERE (Email = @UserAuth OR SDT = @UserAuth) AND Password = @Password AND isActive = 1 AND isDeleted = 0

			END
			ELSE
			BEGIN
				SELECT 401 ErrorCode,dbo.GetErrorMessage(6) AS ErrorMessage

			END
		END
		ELSE
		BEGIN
			SELECT 400 ErrorCode,dbo.GetErrorMessage(5) AS ErrorMessage
			GOTO THOAT
		END
	END
	
	IF @Type = 'Auth-Register'
	BEGIN
		DECLARE @GETDATENOW DATETIME = GETDATE()

		BEGIN
			SELECT 400 ErrorCode,dbo.GetErrorMessage(7) AS ErrorMessage
			GOTO THOAT
		END
		ELSE
		BEGIN

			SELECT 0 ErrorCode,dbo.GetErrorMessage(8) AS ErrorMessage
			GOTO THOAT
		END

	IF @Type = 'Get-Data'

			SELECT 400 AS ErrorCode, dbo.GetErrorMessage(9) AS ErrorMessage

		GOTO THOAT
	END

	IF @Type = 'UpdateActive'

			UPDATE VaccineCategory.dbo.Account

			WHERE ID = @ID

			GOTO THOAT
		END

	END

	END

		IF (SELECT COUNT(*) FROM VaccineCategory.dbo.Account WHERE ID = @ID AND Password = @Password) >= 1

	END

CREATE PROC [dbo].[ExecAgeItem]
    @Type varchar(50) = 'GET-AGE-ITEM',
	@ID int = 1,
	@NameYearOld nvarchar(255) = N'Việt Nam',
	@IsActive bit = 1
AS
BEGIN
	IF @Type NOT IN ('INSERT-AGE','GET-AGE-ITEM','EDITAGE-BYID','ACTIVE-AGE','DELETE-AGE','GET-AGE-PUBLIC')
	BEGIN
		select dbo.GetErrorMessage(1) AS ErrorMessage
		goto THOAT
	END
	DECLARE @Sql nvarchar(255) = ''
	IF @Type = 'INSERT-AGE'
	BEGIN
		BEGIN TRY
			IF EXISTS (
				SELECT 1 
				FROM VaccineCategory.dbo.YeahOldItem 
				WHERE NameYearOld = @NameYearOld
			)
			BEGIN
				IF (SELECT COUNT(*) FROM VaccineCategory.dbo.YeahOldItem WHERE isDeleted = 1 AND @NameYearOld = @NameYearOld) >= 1
				BEGIN
					UPDATE VaccineCategory.dbo.YeahOldItem
					SET isDeleted = 0
					WHERE NameYearOld = @NameYearOld;
				END
				SELECT 400 ErrorCode, dbo.GetErrorMessage(2) ErrorMessage;
				GOTO THOAT
			END
			ELSE
			BEGIN
				DECLARE @GETDATENOW DATETIME = DATEADD(HOUR, 7, GETUTCDATE());

				INSERT INTO VaccineCategory.dbo.YeahOldItem (NameYearOld, CreateDate) 
				VALUES (@NameYearOld, @GETDATENOW);

				SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
				GOTO THOAT
			END
		END TRY
		BEGIN CATCH
			SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
			GOTO THOAT
		END CATCH
	END

	IF @Type  = 'GET-AGE-ITEM'
	BEGIN
		SELECT *,dbo.GetErrorMessage(0) ErrorMessage FROM VaccineCategory.dbo.YeahOldItem WHERE isDeleted = 0
		GOTO THOAT
	END

	IF @Type  = 'GET-AGE-PUBLIC'
	BEGIN
		SELECT ID,NameYearOld,CreateDate,dbo.GetErrorMessage(0) ErrorMessage FROM VaccineCategory.dbo.YeahOldItem WHERE isActive = 1 AND isDeleted = 0
		GOTO THOAT
	END

	IF @Type = 'EDITAGE-BYID'
	BEGIN
		BEGIN TRY
			IF NOT EXISTS (
				SELECT 1 
				FROM VaccineCategory.dbo.YeahOldItem 
				WHERE ID = @ID
			)
			BEGIN
				SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
				GOTO THOAT
			END
			ELSE IF EXISTS (
				SELECT 1 
				FROM VaccineCategory.dbo.YeahOldItem 
				WHERE NameYearOld = @NameYearOld AND ID <> @ID
			)
			BEGIN
				SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
				GOTO THOAT
			END
			ELSE
			BEGIN
				UPDATE VaccineCategory.dbo.YeahOldItem
				SET 
					NameYearOld = @NameYearOld
				WHERE ID = @ID;

				SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
				GOTO THOAT
			END
		END TRY
		BEGIN CATCH
			SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
			GOTO THOAT
		END CATCH
	END

	IF @Type = 'ACTIVE-AGE'
	BEGIN
		IF @IsActive = 1
		BEGIN
			UPDATE dbo.YeahOldItem 
			SET isActive = 1
			WHERE ID = @ID
			SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
			GOTO THOAT
		END
		IF @IsActive = 0
		BEGIN
			UPDATE dbo.YeahOldItem 
			SET isActive = 0
			WHERE ID = @ID
			SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
			GOTO THOAT
		END

	END

	IF @Type = 'DELETE-AGE'

		BEGIN TRY
			UPDATE VaccineCategory.dbo.YeahOldItem
			SET isDeleted = 1
			WHERE ID = @ID;

			BEGIN
				SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
				GOTO THOAT
			END
			ELSE
			BEGIN
				SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
				GOTO THOAT
			END
		END TRY
		BEGIN CATCH
			SELECT ERROR_NUMBER() ErrorCode, ERROR_MESSAGE() ErrorMessage;
			GOTO THOAT
		END CATCH
	END

END

CREATE PROC [dbo].[ExecBooking]
    @Type varchar(50) = 'SCAN-DAILY',
	@Name nvarchar(255) = '',
	@Birthday DATE = '',
	@SDTInject nvarchar(15) = '',
	@NameContact nvarchar(255)= '',
	@DateInject DATE = '',
	@SDT nvarchar(15) = '',
	@Email nvarchar(100) = '',
	@IDUser INT = 21,
	@IDCenter INT ='',
	@vaccinepack nvarchar(max) = '',
	@vaccinesingle nvarchar(max) = '',
	@typyvaccine nvarchar(50) = '',
	@BookID int = 17,
	@NewStatus nvarchar(50) = 'CONFIRMED',
	@ID INT = 17
AS
BEGIN
	IF @Type NOT IN ('BOOKING','GET-DATA-CHART','GET-DATA-BY-BOOKINGID','GET-DATA-BY-ID-USER','UPDATE-STATUS','SCAN-DAILY')
	BEGIN
		select dbo.GetErrorMessage(1) AS ErrorMessage
		goto THOAT
	END
	DECLARE @Sql nvarchar(255) = ''
	--SELECT * FROM VaccineCategory.dbo.BookingInjection


	IF @Type = 'BOOKING'
	BEGIN
		IF @Name IS NULL OR @Name = '' 
			OR @Birthday IS NULL 
			OR @NameContact IS NULL OR @NameContact = '' 
			OR @DateInject IS NULL 
			OR @SDT IS NULL OR @SDT = '' 
			OR @Email IS NULL OR @Email = '' 
			OR @IDUser IS NULL 
			OR @IDCenter IS NULL
		BEGIN
			SELECT 400 ErrorCode, dbo.GetErrorMessage(11) ErrorMessage;
			GOTO THOAT
		END

		BEGIN TRY
			DECLARE @GETDATENOW DATE = GETDATE();

			INSERT INTO VaccineCategory.dbo.BookingInjection(Name, Birthday, SDTInject, NameContact, DateInject, SDT, Email, IDUser, IDCenter)
			VALUES(@Name, @Birthday, @SDTInject, @NameContact, @DateInject, @SDT, @Email, @IDUser, @IDCenter);

			DECLARE @BookingID INT = SCOPE_IDENTITY();

			INSERT INTO VaccineCategory.dbo.BookingInjectionDetail (IDBooking, Name, Price, typevaccine)
			SELECT 
				@BookingID AS IDBooking,
				JSON_VALUE(value, '$.Name') AS Name,
				JSON_VALUE(value, '$.Price') AS Price,
				'PACK'
			FROM OPENJSON(@vaccinepack);

			INSERT INTO VaccineCategory.dbo.BookingInjectionDetail (IDBooking, Name, Price, typevaccine)
			SELECT 
				@BookingID AS IDBooking,
				JSON_VALUE(value, '$.Name') AS Name,
				JSON_VALUE(value, '$.Price') AS Price,
				'SINGLE'
			FROM OPENJSON(@vaccinesingle);

			SELECT *, 
				   @BookingID BookingID, 
				   @Email Email, 
				   0 ErrorCode, 
				   dbo.GetErrorMessage(0) ErrorMessage, 
				   CONVERT(VARCHAR, DateInject, 105) AS FormattedDateInject
			FROM BookingInjection
			WHERE ID = @BookingID;


			GOTO THOAT;
		END TRY
		BEGIN CATCH
			-- Xử lý lỗi
			SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
		END CATCH
	END

	IF @TYPE = 'GET-DATA-CHART'
	BEGIN
		SELECT 
			CAST(CreateDate AS DATE) AS DateOnly, 
			COUNT(*) AS TotalBookings
		FROM VaccineCategory.dbo.BookingInjection
		GROUP BY CAST(CreateDate AS DATE) 
		ORDER BY DateOnly;
	END

	IF @TYPE = 'GET-DATA-BY-BOOKINGID'
	BEGIN
		SELECT 
			BI.ID,
			BI.Name,
			BI.Birthday,
			BI.SDTInject,
			BI.NameContact,
			BI.DateInject,
			BI.IDUser,
			BI.IDCenter,
			BI.SDT AS SDT,
			BI.status,
			VC.Name as NameCenter,
			STRING_AGG(BID.Name, ', ') AS VaccineNames, 
			STRING_AGG(CAST(BID.Price AS VARCHAR), ', ') AS VaccinePrices, 
			STRING_AGG(BID.TypeVaccine, ', ') AS VaccineTypes,
			CONCAT(VC.Address, ', ', VC.Ward, ', ', VC.District, ', ', VC.City) AS Address
		FROM 
			BookingInjection BI
		LEFT JOIN 
			BookingInjectionDetail BID ON BI.ID = BID.IDBooking
		LEFT JOIN 
			VaccineCenter VC ON BI.IDCenter = VC.ID
		GROUP BY
			BI.ID, BI.Name, BI.Birthday, BI.SDTInject, BI.NameContact, 
			BI.DateInject, BI.SDT, BI.IDUser, BI.IDCenter, BI.status, BI.CreateDate,BI.CreateDate, VC.Address, VC.Ward, VC.District, VC.City,VC.Name
		ORDER BY 
			BI.CreateDate DESC;


	END
	
	IF @TYPE = 'GET-DATA-BY-ID-USER'
	BEGIN
		SELECT 
			BI.ID,
			BI.Name,
			BI.Birthday,
			BI.SDTInject,
			BI.NameContact,
			BI.DateInject,
			BI.IDUser,
			BI.IDCenter,
			BI.SDT AS SDT,
			BI.status,
			VC.Name as NameCenter,
			STRING_AGG(BID.Name, ', ') AS VaccineNames,
			STRING_AGG(CAST(BID.Price AS VARCHAR), ', ') AS VaccinePrices,
			STRING_AGG(BID.TypeVaccine, ', ') AS VaccineTypes,
			
			CONCAT(VC.Address, ', ', VC.Ward, ', ', VC.District, ', ', VC.City) AS Address
		FROM 
			BookingInjection BI
		LEFT JOIN 
			BookingInjectionDetail BID ON BI.ID = BID.IDBooking
		LEFT JOIN 
			VaccineCenter VC ON BI.IDCenter = VC.ID 
		WHERE 
			BI.IDUser = @IDUser
		GROUP BY
			BI.ID, BI.Name, BI.Birthday, BI.SDTInject, BI.NameContact, 
			BI.DateInject, BI.SDT, BI.IDUser, BI.IDCenter, BI.status, BI.CreateDate, VC.Address, VC.Ward, VC.District, VC.City,VC.Name
		ORDER BY 
			BI.CreateDate DESC;
		GOTO THOAT
	END
	
	IF @TYPE = 'UPDATE-STATUS'
	BEGIN
		BEGIN TRY
			UPDATE VaccineCategory.dbo.BookingInjection
			SET status = @NewStatus
			WHERE ID =  @ID
			SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage
		END TRY
		BEGIN CATCH
		END CATCH

	END

	IF @TYPE = 'SCAN-DAILY'
	BEGIN
		-- 1. Cập nhật các bản ghi đã quá hạn thành 'EXPIRED' theo giờ Việt Nam
		UPDATE BookingInjection
		SET Status = 'EXPIRED'
		WHERE DateInject < DATEADD(DAY, -1, DATEADD(HOUR, 7, GETUTCDATE())) AND Status != 'EXPIRED' AND status !='CANCELED';


		-- 2. Lấy các bản ghi có DateInject là ngày hôm nay theo giờ Việt Nam
		SELECT 
			b.*, 
			CONVERT(VARCHAR, b.DateInject, 105) AS FormattedDateInject,
			CONCAT(c.Address, ', ',c.Ward, ', ', c.District, ', ', c.City) AS Address,
			c.Name as namecenter
		FROM 
			BookingInjection b
		JOIN 
			VaccineCenter c ON b.IDCenter = c.ID
		WHERE 
			CAST(b.DateInject AS DATE) = CAST(DATEADD(HOUR, 7, GETUTCDATE()) AS DATE);


	END


END

			SELECT 400 ErrorCode, dbo.GetErrorMessage(9) ErrorMessage;
			GOTO THOAT

			GOTO THOAT

			SET isActive = 0
			WHERE ID = @ID

			BEGIN

				GOTO THOAT
			END
			ELSE
			BEGIN
				SELECT 0 ErrorCode, dbo.GetErrorMessage(0) ErrorMessage;
				GOTO THOAT
			END
		END TRY
		BEGIN CATCH
			SELECT ERROR_NUMBER() ErrorCode, ERROR_MESSAGE() ErrorMessage;
			GOTO THOAT
		END CATCH
	END

	@SearchKeyword nvarchar(MAX) = N'Vắc xin' 
AS
BEGIN
	IF @Type NOT IN ('GET','GET-ALL','ADD-POST','DELETE','UPDATE-BY-ID','GET-BY-LINKROUTE','GET-POST-BYID','GET-POST-NAMETYPE','RANDOM_POST','SEARCH','RANDOM_POST_ALL')
	BEGIN
		select dbo.GetErrorMessage(1) AS ErrorMessage
		goto THOAT
	END


	DECLARE @Sql NVARCHAR(MAX) = ''
	IF @Type = 'GET'
	BEGIN
		SET @Sql = 'SELECT ID, Title, ShortContent, LinkImages, CreateDate, LinkRoute FROM Post
					WHERE IsActive = 1 AND IsDeleted = 0
		'
		EXEC(@Sql)
		GOTO THOAT
	END


	IF @Type = 'GET-ALL'
	BEGIN
		SET @Sql = 'SELECT ID, Title, * ,dbo.GetErrorMessage(0) AS ErrorMessage FROM Post
					WHERE IsActive = 1 AND IsDeleted = 0
		'
		EXEC(@Sql)
		GOTO THOAT
	END


	IF @Type = 'ADD-POST'
	BEGIN
		declare @GetDateNow datetime = GETDATE()
		INSERT INTO Post (Title, ShortContent, FullContentDesktop,FullContentMobile, LinkImages, CreateDate, LinkRoute, PostType)
		VALUES (@Title, @ShortContent, @FullContentDesktop,@FullContentMobile, @LinkImages, @GetDateNow, @LinkRoute, @PostType);
		exec(@sql)
		SELECT dbo.GetErrorMessage(0) AS ErrorMessage
		GOTO THOAT
	END


	IF @Type = 'DELETE'
	BEGIN
		IF NOT EXISTS (SELECT 1 FROM VaccineCategory.dbo.Post WHERE ID = @ID)
		BEGIN
			SELECT @ID ID, dbo.GetErrorMessage(3) AS ErrorMessage
			GOTO THOAT
		END
		ELSE
		BEGIN
			UPDATE VaccineCategory.dbo.Post
			SET IsDeleted = 1
			WHERE ID = @ID;
			SELECT dbo.GetErrorMessage(0) AS ErrorMessage
			GOTO THOAT
		END
	END

	IF @Type = 'UPDATE-BY-ID'
	BEGIN
		IF NOT EXISTS (SELECT 1 FROM VaccineCategory.dbo.Post WHERE ID = @ID)
		BEGIN
			SELECT dbo.GetErrorMessage(3) AS ErrorMessage
			GOTO THOAT
		END
		ELSE
		BEGIN
			UPDATE VaccineCategory.dbo.Post
			SET Title = @Title,
				ShortContent = @ShortContent,
				FullContentDesktop = @FullContentDesktop,
				FullContentMobile = @FullContentMobile,
				LinkImages = @LinkImages,
				LinkRoute = @LinkRoute,
				PostType = @PostType
			WHERE ID = @ID;
			SELECT dbo.GetErrorMessage(0) AS ErrorMessage
			GOTO THOAT
		END
	END

	IF @Type = 'GET-BY-LINKROUTE'
	BEGIN
		if (select count(*) from VaccineCategory.dbo.Post where LinkRoute = @LinkRoute) <= 0
		begin
			SELECT 404 ErrorCode,dbo.GetErrorMessage(3) AS ErrorMessage
			goto thoat
		end
		else
		begin
			select TOP(1) *,0 ErrorCode, dbo.GetErrorMessage(0) AS ErrorMessage
			from VaccineCategory.dbo.Post
			where IsDeleted = 0 AND IsActive = 1 AND LinkRoute = @LinkRoute
			GOTO THOAT
		end
	END

	IF @Type = 'GET-POST-BYID'
	BEGIN
		SELECT TOP (1) [ID],[Title],[ShortContent],[LinkImages],[LinkRoute],[PostType]
		FROM [VaccineCategory].[dbo].[Post]
		WHERE ID = @ID

	IF @Type = 'GET-POST-NAMETYPE'

					FROM Post
					WHERE IsActive = 1 AND IsDeleted = 0
					ORDER BY CreateDate DESC'
		EXEC(@Sql)
		GOTO THOAT
	END

	IF @Type = 'SEARCH'
	BEGIN
		SELECT Title, LinkRoute, LinkImages, ShortContent
		FROM VaccineCategory.dbo.Post
		WHERE (Title LIKE '%' + @SearchKeyword + '%' 
			   OR ShortContent LIKE '%' + @SearchKeyword + '%')
		AND IsActive = 1
		AND IsDeleted = 0
	END

CREATE PROCEDURE [dbo].[InsertMessage]
	@Type Nvarchar(255) = 'GET-ADMIN-ALL',
    @Sender NVARCHAR(100) = '0338988152',
    @Receiver NVARCHAR(100) = 'admin@manhduc',
    @MessageContent NVARCHAR(500) = NULL
AS
BEGIN
	IF @Type = 'INSERT-MESSAGE'
	BEGIN
		INSERT INTO Messages (Sender, Receiver, MessageContent, Timestamp)
        VALUES (@Sender, @Receiver, @MessageContent, GETDATE());
		SELECT 0 ErrorCode
	END

	IF @Type = 'GET'
    BEGIN
        SELECT 
            Sender, 
            Receiver, 
            MessageContent, 
            Timestamp
        FROM 
            Messages
        WHERE 
            (Sender = @Sender AND Receiver = @Receiver)
            OR (Sender = @Receiver AND Receiver = @Sender)
        ORDER BY 
            Timestamp ASC; 
    END

	IF @Type = 'GET-ADMIN-ALL'
	BEGIN
		 ;WITH RankedMessages AS (
        SELECT 
            Sender,
            Receiver,
            MessageContent,
            Timestamp,
            ROW_NUMBER() OVER (PARTITION BY Sender ORDER BY Timestamp DESC) AS RowNum
        FROM Messages
        WHERE Receiver = @Receiver
		)
		SELECT Sender, Receiver, MessageContent, Timestamp
		FROM RankedMessages
		WHERE RowNum = 1
		ORDER BY Timestamp DESC;
	END



END
